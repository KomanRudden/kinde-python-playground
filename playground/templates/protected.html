Using the integration directly

Without the playground's guards, protect a route with the framework integration alone:

{% if framework == "flask" %} 
@app.route("/protected/native")
def native():
    if not oauth.is_authenticated():
        return redirect("/login")
    return {"user": oauth.get_user_info()["email"]}
{% else %} 
def current_user(request: Request) -> dict:
    if not oauth.is_authenticated():
        raise HTTPException(status_code=401)
    return oauth.get_user_info()

@app.get("/protected/native")
async def native(user: dict = Depends(current_user)):
    return {"user": user["email"]}
